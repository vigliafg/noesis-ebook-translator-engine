# Noesis Ebook Translator — Engine (CLI + TUI)
#
# The single-file engine needs no GUI toolkit: these are the runtime
# dependencies of the inlined Ebook Translator engine.

# EPUB reading/writing (the translation pipeline core).
ebooklib

# XHTML parsing/rewriting that preserves inline markup, links and the TOC.
lxml

# HTTP stack used by the verbatim Calibre-plugin translation engines.
mechanize

# CSS selector engine used by the element handler to match XHTML nodes.
cssselect

# Optional: SOCKS5 proxy support. The engine imports `socks` lazily and only
# when a SOCKS5 proxy is configured; uncomment (or `pip install PySocks`) to
# route the translation engines through a SOCKS5 proxy.
# PySocks
